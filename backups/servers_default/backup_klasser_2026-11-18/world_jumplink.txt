function onActionServerSide()
{
  if (params[0] == "Warp")
  {
    with(findplayer(params[1]))
    {
      setlevel2(params[2][0],params[2][1],params[2][2]); 
    } 
  }
  if (params[0] == "Destroy")
  {
    echo("Script: this.link not found on the clientside of world_jumplink in " @ this.level @ " X:" @ this.x @ " Y:" @ this.y);
    this.destroy();
  }
}

//#CLIENTSIDE
function onCreated()
{
  CheckForConfig();
  Config(); 
}

function onPlayerenters()
{
  CheckForConfig();
  Config(); 
}

function CheckForConfig()
{
  if (this.link == NULL)
    triggeraction(this.x,this.y,"ServerSide","Destroy");
    
  if (this.bounds == NULL)
    this.bounds = {2,2};
    
  //Another settable variable is this.TouchJump
  //this.TouchJump = #;
  //This enables the player to jump that number of units when
  //  touching the jumplink.
  //# can be a number or "z","x","c", with quotes, to call their unique jumps.
}

function Config()
{
  setshape(1,16*this.bounds[0],16*this.bounds[1]);
  
  this.strength = .25;
  this.loopSpeed = .05;

  setTimer(this.loopSpeed);
}

function onTimeout()
{
  if (InBounds() == true)
  {
    if (this.TouchJump != NULL)
      Jump.ForceJump(this.TouchJump);
      
    if (player.z > 0)
    {
      this.jumping = true;
    
      dontblock();
     
      if (player.x < (this.x - 1.5) + this.width/2)
        player.x+=this.strength;
      if (player.x > (this.x - 1.5) + this.width/2)
        player.x-=this.strength;
      if (player.y < (this.y - 2.75) + this.height/2)
        player.y+=this.strength;
      if (player.y > (this.y - 2.75) + this.height/2)
        player.y-=this.strength;
    }
    else if (this.jumping == true)
    {
      this.jumping = false;
    
      blockagain();
      
      temp.ACC = player.account;
      triggeraction(this.x,this.y,"ServerSide","Warp",ACC,this.link);
    }
  }
  else
  {
    this.jumping = false;
    blockagain();
  }
  setTimer(this.loopSpeed);
}

function InBounds()
{
  if (player.x in |(this.x-2.5),(this.x)+this.width| && player.y in |(this.y-3.5),(this.y-1)+this.height|)
      return true;
  else
      return false;
}
