function onPlayerTouchsMe() {

function onTimeout() {
  if (playerattached) {
    player.x = x - .5;
    player.dir = 0;
    if ((player.y <= y-2 && keydown(0)) || (player.y >= y+2 && keydown(2)))
      detachplayer();
    freezeplayer(0.05);
    if (keydown(0))
      player.y -= 0.2;
    else if (keydown(2))
      player.y += 0.2;
  }
  setTimer(0.05);
}
