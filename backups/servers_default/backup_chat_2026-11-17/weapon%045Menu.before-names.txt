function toggle() {

Menu_Window.hide();
} else {
if (client.winext == "off") {
Menu_Window.position = {int((screenwidth - 330) / 2), int((screenheight - 330) / 2)};
}
this.showPage(this.page);
Menu_Window.show();
}
}

function onKeyPressed(code, key) {

}
}

function onoff(v) {

return "On";
}

}

function refreshSound() {
Mn_VFx.value = $pref::audio::sfxvolume;
Mn_VMid.value = $pref::audio::midivolume;
Mn_VMp3.value = $pref::audio::mp3volume;
Mn_VRad.value = $pref::audio::radiovolume;
Mn_VFxV.text = int($pref::audio::sfxvolume) @ "%";
Mn_VMidV.text = int($pref::audio::midivolume) @ "%";
Mn_VMp3V.text = int($pref::audio::mp3volume) @ "%";
Mn_VRadV.text = int($pref::audio::radiovolume) @ "%";
if (client.audiomuted == "yes") {
Mn_Mute.text = "Muted";
} else {
Mn_Mute.text = "Sound on";
}
if ($pref::audio::reversestereo) {
Mn_Rev.text = "Swapped";
} else {
Mn_Rev.text = "Normal";
}
echo("-Menu: volumes " @ $pref::audio::sfxvolume @ "/" @ $pref::audio::midivolume @ "/" @ $pref::audio::mp3volume @ "/" @ $pref::audio::radiovolume @ " stereo " @ $pref::audio::reversestereo);
}

function setVolume(which, v) {
temp.n = int(v);
if (which == 0) {
$pref::audio::sfxvolume = temp.n;
Mn_VFxV.text = temp.n @ "%";
} else if (which == 1) {
$pref::audio::midivolume = temp.n;
Mn_VMidV.text = temp.n @ "%";
} else if (which == 2) {
$pref::audio::mp3volume = temp.n;
Mn_VMp3V.text = temp.n @ "%";
} else {
$pref::audio::radiovolume = temp.n;
Mn_VRadV.text = temp.n @ "%";
}
echo("-Menu: volume " @ which @ " = " @ temp.n);
}

function toggleMute() {
if (client.audiomuted == "yes") {
$pref::audio::sfxvolume = client.premute0;
$pref::audio::midivolume = client.premute1;
$pref::audio::mp3volume = client.premute2;
$pref::audio::radiovolume = client.premute3;
client.audiomuted = "no";
} else {
client.premute0 = $pref::audio::sfxvolume;
client.premute1 = $pref::audio::midivolume;
client.premute2 = $pref::audio::mp3volume;
client.premute3 = $pref::audio::radiovolume;
$pref::audio::sfxvolume = 0;
$pref::audio::midivolume = 0;
$pref::audio::mp3volume = 0;
$pref::audio::radiovolume = 0;
client.audiomuted = "yes";
}
this.refreshSound();
}

function toggleStereo() {
if ($pref::audio::reversestereo) {
$pref::audio::reversestereo = 0;
} else {
$pref::audio::reversestereo = 1;
}
this.refreshSound();
}

function applyFps() {
temp.f = int(client.fpsmode);
if (temp.f > 0) {
adventure_setframetick(temp.f);
}
echo("-Menu: frametick=" @ adventure_getframetick() @ " ms");
}

function fpsMode() {
temp.f = int(client.fpsmode);
if (temp.f > 0) {
return temp.f;
}
return 50;
}

function fpsText() {
return int(1000 / this.fpsMode()) @ " FPS";
}

function cycleFps() {
temp.t = this.fpsMode();
temp.n = 50;
if (temp.t >= 40) {
temp.n = 33;
} else if (temp.t >= 25) {
temp.n = 16;
}
client.fpsmode = "" @ temp.n;
adventure_setframetick(temp.n);
echo("-Menu: frametick set to " @ temp.n @ " ms (getframetick says " @ adventure_getframetick() @ ")");
this.refreshSettings();
}

function refreshSettings() {
if (client.winext == "off") {
Mn_SPlext.text = "In the game";
} else {
Mn_SPlext.text = "Own windows";
}
Mn_SFps.text = this.fpsText();
Mn_SStart.text = this.onoff(client.chatstart == "on");
temp.r = 40;
if (client.chatradius > 0) {
temp.r = client.chatradius;
}
Mn_SRange.text = temp.r @ " tiles";
Mn_SBubble.text = this.onoff(client.chatbubble == 1);
temp.a = 1;
if (client.chatalpha > 0) {
temp.a = client.chatalpha;
}
Mn_SAlpha.text = int(temp.a * 100) @ " %";
}

function refreshInfo() {
Mn_I0.text = "Server: Valikorlia Reborn v7";
Mn_I1.text = "Account: " @ player.account;
Mn_I2.text = "Name: " @ player.nick;
Mn_I3.text = "Guild: " @ player.guild;
Mn_I4.text = "Level: " @ level.name;
Mn_I5.text = "Position: " @ int(player.x) @ ", " @ int(player.y);
Mn_I6.text = "Players here: " @ (players.size() + 1);
Mn_I7.text = "Chat range: " @ this.chatRange() @ " tiles";
}

function chatRange() {
if (client.chatradius > 0) {
return client.chatradius;
}
return 40;
}

function cycleRange() {
temp.r = this.chatRange();
if (temp.r < 20) {
client.chatradius = 20;
} else if (temp.r < 40) {
client.chatradius = 40;
} else if (temp.r < 80) {
client.chatradius = 80;
} else {
client.chatradius = 10;
}
this.refreshSettings();
}

function cycleAlpha() {
temp.a = 1;
if (client.chatalpha > 0) {
temp.a = client.chatalpha;
}
if (temp.a > 0.9) {
client.chatalpha = 0.8;
} else if (temp.a > 0.7) {
client.chatalpha = 0.6;
} else if (temp.a > 0.5) {
client.chatalpha = 0.4;
} else {
client.chatalpha = 1;
}
this.refreshSettings();
}

function toggleStart() {
if (client.chatstart == "on") {
client.chatstart = "off";
} else {
client.chatstart = "on";
}
this.refreshSettings();
}

function toggleBubble() {
if (client.chatbubble == 1) {
client.chatbubble = 0;
} else {
client.chatbubble = 1;
}
this.refreshSettings();
}

function resetAll() {
client.chatstart = "off";
client.chatradius = 40;
client.chatbubble = 0;
client.chatalpha = 1;
client.fpsmode = "50";
adventure_setframetick(50);
client.winext = "on";
client.winextgen = int(client.winextgen) + 1;
this.refreshSettings();
}

function checkWinExt() {
if (int(client.winextgen) != this.extgen) {
this.extgen = int(client.winextgen);
temp.was = Menu_Window.visible;
this.build();
if (temp.was) {
this.toggle();
}
}
}

function onTimeOut() {
this.checkWinExt();
if (client.openwindow == "menu") {
client.openwindow = "";
this.toggle();
}
if (client.openwindow == "menuhide") {
client.openwindow = "";
Menu_Window.hide();
}
if (this.page == 3 && Menu_Window.visible) {
this.tick = this.tick + 1;
if (this.tick >= 5) {
this.tick = 0;
this.refreshInfo();
}
}
setTimer(0.2);
}
