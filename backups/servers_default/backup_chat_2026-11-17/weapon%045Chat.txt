function onCreated() {
  echo("-Chat: onCreated");
  this.lines = {};
  this.off = 0;
  this.distance = 40;
  this.pending = false;
  this.pdelay = 0;
  this.open = true;
  if (client.chatstart != "on") {
    this.open = false;
  }
  this.tick = 0;
  this.build();
  this.applySettings();
  this.catchevent("GraalControl", "onResize", "onGraalResize");
  setTimer(0.05);
}
function applySettings() {
  this.distance = 40;
  if (client.chatradius > 0) {
    this.distance = client.chatradius;
  }
  if (client.chatalpha > 0) {
    Chat_Window.alpha = client.chatalpha;
  }
}
function onGraalResize() {
  this.build();
}

function toggle() {
  if (this.open == false) {
    client.openwindow = "menuhide";
    Chat_Window.show();
    this.open = true;
    Chat_Input.makefirstresponder(true);
  } else {
    Chat_Window.hide();
    this.open = false;
    GraalControl.makefirstresponder(true);
  }
}
function onKeyPressed(code, key) {
  if (code == 9)
    this.toggle();
}
function onInputKeyDown(objn, key, code, keycode) {
  echo("-Chat: input key " @ code @ "/" @ keycode);
  if (code == 13 || keycode == 28)
    this.sendInput();
  else if (keycode == 73 || code == 33)
    this.scrollBy(5);
  else if (keycode == 81 || code == 34)
    this.scrollBy(-5);
  else if (keycode == 72 || code == 38)
    this.scrollBy(1);
  else if (keycode == 80 || code == 40)
    this.scrollBy(-1);
}
function onWheelUp() {
  this.scrollBy(1);
}
function onWheelDown() {
  this.scrollBy(-1);
}
function addLine(nick, text) {
  this.lines.add(nick @ ": " @ text);
  if (this.off > 0)
    this.off++;
  while (this.lines.size() > 200)
    this.lines.delete(0);
  this.clampOff();
  this.refresh();
}
function clampOff() {
  temp.max = this.lines.size() - 5;
  if (temp.max < 0)
    temp.max = 0;
  if (this.off > temp.max)
    this.off = temp.max;
  if (this.off < 0)
    this.off = 0;
}
function scrollBy(n) {
  this.off = this.off + n;
  this.clampOff();
  this.refresh();
}
function line(i) {
  temp.k = this.lines.size() - 5 - this.off + i;
  if (temp.k < 0 || temp.k >= this.lines.size())
    return "";
  return this.lines[temp.k];
}
function refresh() {
  Chat_L0.text = this.line(0);
  Chat_L1.text = this.line(1);
  Chat_L2.text = this.line(2);
  Chat_L3.text = this.line(3);
  Chat_L4.text = this.line(4);
  if (this.off > 0)
    Chat_Up.text = "^" @ this.off;
  else
    Chat_Up.text = "^";
}
function checkPlayer(p) {
  if (p.chat == "" || p.chat == this.(@ "c_" @ p.account))
    return;
  this.(@ "c_" @ p.account) = p.chat;
  if (p.chat.starts("/"))
    return;
  this.addLine(p.nick, p.chat);
}
function onTimeOut() {
  temp.pm = "" @ client.pmline;
  if (temp.pm.length() > 0) {
    client.pmline = "";
    this.addLine("PM", temp.pm);
  }
  this.tick = this.tick + 1;
  if (this.tick >= 20) {
    this.tick = 0;
    this.applySettings();
  }
  if (client.openwindow == "chat") {
    client.openwindow = "";
    this.toggle();
  }
  if (this.pending) {
    this.pdelay = this.pdelay - 0.05;
    if (this.pdelay <= 0) {
      player.chat = "";
      this.pending = false;
    }
  } else
    this.checkPlayer(player);
  for (temp.p: players) {
    if (vectordist({player.x, player.y, 0}, {temp.p.x, temp.p.y, 0}) <= this.distance)
      this.checkPlayer(temp.p);
  }
  setTimer(0.05);
}
function sendInput() {
  temp.t = Chat_Input.text;
  if (temp.t == "")
    return;
  Chat_Input.text = "";
  this.(@ "c_" @ player.account) = temp.t;
  if (!temp.t.starts("/"))
    this.addLine(player.nick, temp.t);
  player.chat = temp.t;
  if (client.chatbubble != 1) {
    this.pending = true;
    this.pdelay = 3;   // CHAT_HOLD
  }
}
function Chat_Up.onAction() {
  scrollBy(5);
}
function Chat_Down.onAction() {
  scrollBy(-5);
}
function Chat_Send.onAction() {
  sendInput();
}
